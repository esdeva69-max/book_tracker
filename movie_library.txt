import tkinter as tk
from tkinter import ttk, messagebox
import json
import os

class MovieLibraryApp:
    def __init__(self, root):
        self.root = root
        self.root.title("Movie Library")
        self.movies = []  # Список для хранения фильмов в памяти

        # --- Блок 1: Поля ввода ---
        frame_inputs = ttk.LabelFrame(root, text="Добавить фильм")
        frame_inputs.grid(row=0, column=0, columnspan=2, padx=10, pady=10, sticky="ew")

        ttk.Label(frame_inputs, text="Название:").grid(row=0, column=0, padx=5, pady=2, sticky="e")
        self.title_entry = ttk.Entry(frame_inputs, width=30)
        self.title_entry.grid(row=0, column=1, padx=5, pady=2, sticky="w")

        ttk.Label(frame_inputs, text="Жанр:").grid(row=1, column=0, padx=5, pady=2, sticky="e")
        self.genre_entry = ttk.Entry(frame_inputs, width=30)
        self.genre_entry.grid(row=1, column=1, padx=5, pady=2, sticky="w")

        ttk.Label(frame_inputs, text="Год выпуска:").grid(row=2, column=0, padx=5, pady=2, sticky="e")
        self.year_entry = ttk.Entry(frame_inputs, width=10)
        self.year_entry.grid(row=2, column=1, padx=5, pady=2, sticky="w")

        ttk.Label(frame_inputs, text="Рейтинг (0-10):").grid(row=3, column=0, padx=5, pady=2, sticky="e")
        self.rating_entry = ttk.Entry(frame_inputs, width=10)
        self.rating_entry.grid(row=3, column=1, padx=5, pady=2, sticky="w")

        # Кнопка добавления фильма
        ttk.Button(frame_inputs,
                   text="Добавить фильм",
                   command=self.add_movie).grid(row=4,
                                                columnspan=2,
                                                pady=10)

        # --- Блок 2: Таблица для отображения ---
        self.tree = ttk.Treeview(root,
                                 columns=(1, 2, 3, 4),
                                 show='headings',
                                 height=15)
        self.tree.grid(row=1, column=0, columnspan=2, padx=10)

        self.tree.heading(1, text="Название")
        self.tree.heading(2, text="Жанр")
        self.tree.heading(3, text="Год")
        self.tree.heading(4, text="Рейтинг")

        self.tree.column(1, anchor='w', width=250)
        self.tree.column(2, anchor='w', width=150)
        self.tree.column(3, anchor='center', width=80)
        self.tree.column(4, anchor='center', width=80)

        # --- Блок 3: Фильтрация ---
        frame_filter = ttk.LabelFrame(root, text="Фильтр")
        frame_filter.grid(row=2, column=0,
                          padx=(10, 5), pady=(0, 10), sticky="ew")

        ttk.Label(frame_filter,
                  text="Жанр:").grid(row=0,
                                     column=0,
                                     padx=(5, 0),
                                     pady=2)

        self.genre_filter = ttk.Entry(frame_filter)
        self.genre_filter.grid(row=0,
                               column=1,
                               padx=(0, 5),
                               pady=2)

        ttk.Label(frame_filter,
                  text="Год:").grid(row=1,
                                    column=0,
                                    padx=(5, 0),
                                    pady=(2, 8))

        self.year_filter = ttk.Entry(frame_filter)
        self.year_filter.grid(row=1,
                              column=1,
                              padx=(0, 5),
                              pady=(2, 8))

         # Кнопка фильтра с явным указанием row и column
         ttk.Button(frame_filter,
                    text="Применить фильтр",
                    command=self.apply_filter).grid(rowspan=2,
                                                    row=2,
                                                    columnspan=2,
                                                    padx=(5, 5),
                                                    pady=(8, 8))

         # --- Блок 4: Кнопки сохранения/загрузки ---
         frame_io = ttk.Frame(root)
         frame_io.grid(row=2, column=1)

         ttk.Button(frame_io,
                    text="Сохранить в JSON",
                    command=self.save_to_json).pack(pady=(10, 2))

         ttk.Button(frame_io,
                    text="Загрузить из JSON",
                    command=self.load_from_json).pack(pady=(2, 10))

    # --- Логика добавления фильма ---
    def add_movie(self):
         title = self.title_entry.get().strip()
         genre = self.genre_entry.get().strip()
         year_raw = self.year_entry.get().strip()
         rating_raw = self.rating_entry.get().strip()

         # Валидация
         if not title or not genre or not year_raw or not rating_raw:
             messagebox.showerror("Ошибка", "Все поля должны быть заполнены!")
             return

         if not year_raw.isdigit():
             messagebox.showerror("Ошибка", "Год выпуска должен быть числом!")
             return

         year = int(year_raw)

         if not rating_raw.replace('.', '', 1).isdigit():
             messagebox.showerror("Ошибка", "Рейтинг должен быть числом!")
             return

         rating = float(rating_raw)

         if rating < 0 or rating > 10:
             messagebox.showerror("Ошибка", "Рейтинг должен быть от 0 до 10!")
             return

         # Добавление в таблицу и в список в памяти
         self.tree.insert("", tk.END,
                          values=(title[:47] + '...' if len(title) > 50 else title,
                                  genre[:47] + '...' if len(genre) > 50 else genre,
                                  year,
                                  f"{rating:.1f}"))

         self.movies.append({"title": title,
                            "genre": genre,
                            "year": year,
                            "rating": rating})

         # Очистка полей
         self.title_entry.delete(0, tk.END)
         self.genre_entry.delete(0, tk.END)
         self.year_entry.delete(0, tk.END)
         self.rating_entry.delete(0, tk.END)

    # --- Логика фильтрации ---
    def apply_filter(self):
         genre_query = self.genre_filter.get().strip().lower()
         year_query_raw = self.year_filter.get().strip()
         
         try:
             min_year = int(year_query_raw) if year_query_raw else -1 # -1 значит "не фильтровать"
         except ValueError:
             messagebox.showerror("Ошибка", "В фильтре года должно быть число!")
             return

         for child in self.tree.get_children():
             values = self.tree.item(child)['values']
             # Поиск полной записи по названию (первый элемент кортежа values)
             movie_db = next((m for m in self.movies if m['title'] == values[0]), None)

             if movie_db is None:
                 continue

             match_genre = (genre_query == "") or (genre_query in movie_db['genre'].lower())
             match_year = (min_year == -1) or (movie_db['year'] >= min_year)

             if match_genre and match_year:
                 self.tree.item(child , tags='') # Показать строку (убрать теги скрытия)
             else:
                 self.tree.item(child , tags='hidden') # Скрыть строку

         self.tree.tag_configure('hidden', elide=True) # Настройка тега для скрытия

    # --- Логика сохранения в JSON ---
    def save_to_json(self):
         if not self.movies:
             messagebox.showinfo("Инфо", "Нет фильмов для сохранения.")
             return

         try:
             with open('movies.json', 'w', encoding='utf-8') as f:
                 json.dump(self.movies, f, ensure_ascii=False, indent=4)
             messagebox.showinfo("Успех", "Данные сохранены в movies.json")
         except Exception as e:
             messagebox.showerror("Ошибка", f"Не удалось сохранить файл: {e}")

    # --- Логика загрузки из JSON ---
    def load_from_json(self):
         try:
             if not os.path.exists('movies.json'):
                 messagebox.showinfo("Инфо", "Файл movies.json не найден.")
                 return

             with open('movies.json', 'r', encoding='utf-8') as f:
                 self.movies = json.load(f)

             # Очистка текущей таблицы перед загрузкой
             for item in self.tree.get_children():
                 self.tree.delete(item)
                 
             # Заполнение таблицы из загруженных данных
             for movie in self.movies:
                 self.tree.insert("", tk.END,
                     values=(movie['title'][:47] + '...' if len(movie['title']) > 50 else movie['title'],
                             movie['genre'][:47] + '...' if len(movie['genre']) > 50 else movie['genre'],
                             movie['year'],
                             f"{movie['rating']:.1f}"))
             
             messagebox.showinfo("Успех", "Данные загружены из movies.json")
             
         except json.JSONDecodeError:
             messagebox.showerror("Ошибка", "Файл movies.json повреждён или имеет неверный формат.")
         except Exception as e:
             messagebox.showerror("Ошибка", f"Не удалось загрузить файл: {e}")

if __name__ == "__main__":
    root = tk.Tk()
    app = MovieLibraryApp(root)
    root.mainloop()
